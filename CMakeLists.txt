cmake_minimum_required(VERSION 3.16)
project(QtWidgetsLearningPath VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets)
include(CTest)

# 默认构建全部；也可 -DQT_LESSON=07_06_text_editor 只构建一个项目。
set(QT_LESSON "" CACHE STRING "Single example directory name, or empty for all")
file(GLOB lesson_directories LIST_DIRECTORIES true "${CMAKE_CURRENT_SOURCE_DIR}/examples/[0-9][0-9]_*")
list(SORT lesson_directories)
foreach(lesson_dir IN LISTS lesson_directories)
    if(NOT IS_DIRECTORY "${lesson_dir}")
        continue()
    endif()
    get_filename_component(lesson "${lesson_dir}" NAME)
    if(QT_LESSON AND NOT lesson STREQUAL QT_LESSON)
        continue()
    endif()
    file(GLOB lesson_sources CONFIGURE_DEPENDS
        "${lesson_dir}/*.cpp" "${lesson_dir}/*.h" "${lesson_dir}/*.ui" "${lesson_dir}/*.qrc")
    qt_add_executable(${lesson} WIN32 ${lesson_sources})
    target_link_libraries(${lesson} PRIVATE Qt6::Widgets)
    # uic 生成的头文件位于构建目录；提升控件的本地头文件位于项目源码目录。
    target_include_directories(${lesson} PRIVATE "${lesson_dir}")
    target_compile_definitions(${lesson} PRIVATE QT_LESSON_SOURCE_DIR="${lesson_dir}")
    set_target_properties(${lesson} PROPERTIES AUTOUIC_SEARCH_PATHS "${lesson_dir}")
    if(MSVC)
        target_compile_options(${lesson} PRIVATE /utf-8)
    endif()
    if(BUILD_TESTING)
        add_test(NAME ${lesson}
            COMMAND $<TARGET_FILE:${lesson}> --verify "${CMAKE_BINARY_DIR}/verification-output/${lesson}")
        if(WIN32)
            # Windows offscreen 后端未必能找到系统字体，使用本机窗口后端。
            set(test_environment "QT_FORCE_STDERR_LOGGING=1")
        else()
            set(test_environment "QT_QPA_PLATFORM=offscreen;QT_FORCE_STDERR_LOGGING=1")
        endif()
        set_tests_properties(${lesson} PROPERTIES TIMEOUT 30
            ENVIRONMENT "${test_environment}")
    endif()
endforeach()
if(QT_LESSON AND NOT TARGET ${QT_LESSON})
    message(FATAL_ERROR "Unknown QT_LESSON: ${QT_LESSON}")
endif()
